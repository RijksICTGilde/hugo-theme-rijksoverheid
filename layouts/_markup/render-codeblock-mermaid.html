{{- /* Mermaid-diagram als voorgerenderde SVG's. Contract met het renderscript van de site: het theme rendert zelf niets. Voor de build maakt de site van elk mermaid-blok twee SVG's, een voor het lichte en een voor het donkere thema (de download is de lichte): static/images/render//-light.svg static/images/render//-dark.svg Het pad is relatief aan de projectmap (of de map van een module). Ontbreekt een van de twee, dan stopt de build met beide verwachte paden. De lichte SVG heeft een viewBox nodig: daaruit komen width en height tegen verspringen. subpad het pad van het contentbestand ten opzichte van content/, zonder extensie; bij index.md en _index.md alleen de map. content/docs/intro.md -> docs/intro content/docs/opzet/index.md -> docs/opzet slug uit accTitle: kleine letters, witruimte wordt "-", alles behalve a-z, 0-9 en "-" vervalt, reeksen "-" worden er een, geen "-" aan begin of eind. Dezelfde slug voor de tweede keer op een pagina wordt -2, dan -3. Zonder accTitle: diagram-, met n het volgnummer van het blok onder alle mermaid-blokken op de pagina (met en zonder accTitle), vanaf 1. Tekstalternatief: accTitle wordt de alt, accDescr (op een regel of als accDescr { ... }) een verborgen beschrijving na de afbeeldingen. Zonder accTitle is accDescr de alt; zonder beide waarschuwt de build. */ -}} {{- $inner := .Inner -}} {{- $accTitle := "" -}} {{- $accDescr := "" -}} {{- $inMultiDescr := false -}} {{- range split $inner "\n" -}} {{- $line := strings.TrimSpace . -}} {{- if $inMultiDescr -}} {{- if strings.Contains $line "}" -}} {{- $inMultiDescr = false -}} {{- else if $line -}} {{- if $accDescr -}} {{- $accDescr = printf "%s %s" $accDescr $line -}} {{- else -}} {{- $accDescr = $line -}} {{- end -}} {{- end -}} {{- else if and (hasPrefix $line "accTitle:") (not $accTitle) -}} {{- /* De eerste accTitle telt, net als in het renderscript. */ -}} {{- $accTitle = strings.TrimSpace (strings.TrimPrefix "accTitle:" $line) -}} {{- else if hasPrefix $line "accDescr {" -}} {{- $inMultiDescr = true -}} {{- else if hasPrefix $line "accDescr:" -}} {{- $accDescr = strings.TrimSpace (strings.TrimPrefix "accDescr:" $line) -}} {{- end -}} {{- end -}} {{- /* LogicalName en niet ContentBaseName: die geeft bij een bundle de mapnaam. */ -}} {{- $pageDir := path.Dir .Page.File.Path -}} {{- $baseName := replaceRE `\.[^.]+$` "" .Page.File.LogicalName -}} {{- $subdir := $pageDir -}} {{- if and (ne $baseName "index") (ne $baseName "_index") -}} {{- $subdir = path.Join $pageDir $baseName -}} {{- end -}} {{- /* Nummer en slug per blok bewaren onder zijn Ordinal: rendert Hugo de inhoud nog een keer (een ander uitvoerformaat), dan krijgt hetzelfde blok hetzelfde nummer in plaats van een doorgeteld nummer. */ -}} {{- $blokken := .Page.Store.Get "mermaid-blokken" | default dict -}} {{- $sleutel := printf "b%d" .Ordinal -}} {{- $blok := index $blokken $sleutel -}} {{- if not $blok -}} {{- $nummer := add (len $blokken) 1 -}} {{- $basis := "" -}} {{- $slug := "" -}} {{- if $accTitle -}} {{- /* \s plus \v, \p{Z} en U+FEFF is de witruimte van \s in JavaScript. */ -}} {{- $basis = $accTitle | lower | replaceRE `[\s\v\p{Z}\x{FEFF}]+` "-" | replaceRE `[^a-z0-9-]` "" | replaceRE `-+` "-" | replaceRE `^-|-$` "" -}} {{- $eerder := 0 -}} {{- range $blokken -}} {{- if eq .basis $basis -}}{{ $eerder = add $eerder 1 }}{{- end -}} {{- end -}} {{- $slug = cond (gt $eerder 0) (printf "%s-%d" $basis (add $eerder 1)) $basis -}} {{- else -}} {{- $slug = printf "diagram-%d" $nummer -}} {{- end -}} {{- $blok = dict "nummer" $nummer "basis" $basis "slug" $slug -}} {{- .Page.Store.Set "mermaid-blokken" (merge $blokken (dict $sleutel $blok)) -}} {{- end -}} {{- $slug := $blok.slug -}} {{- $lightPad := path.Join "images/render" $subdir (printf "%s-light.svg" $slug) -}} {{- $darkPad := path.Join "images/render" $subdir (printf "%s-dark.svg" $slug) -}} {{- $lightFile := path.Join "static" $lightPad -}} {{- $darkFile := path.Join "static" $darkPad -}} {{- if and (os.FileExists $lightFile) (os.FileExists $darkFile) -}} {{- /* De bestandsnaam verandert niet bij een nieuwe render; de inhoudshash in de URL voorkomt dat browser of CDN een oude SVG blijft tonen. */ -}} {{- $lightSVG := os.ReadFile $lightFile -}} {{- $lightUrl := printf "%s?v=%s" $lightPad (substr ($lightSVG | md5) 0 8) | relURL -}} {{- $darkUrl := printf "%s?v=%s" $darkPad (substr (os.ReadFile $darkFile | md5) 0 8) | relURL -}} {{- $viewBox := findRESubmatch `viewBox="[-\d.]+[ ,]+[-\d.]+[ ,]+([\d.]+)[ ,]+([\d.]+)"` $lightSVG 1 -}} {{- if not $viewBox -}} {{- errorf "Diagram '%s' op pagina %s heeft geen leesbare viewBox." $slug .Page.File.Path -}} {{- end -}} {{- $breedte := int (math.Ceil (index (index $viewBox 0) 1)) -}} {{- $hoogte := int (math.Ceil (index (index $viewBox 0) 2)) -}} {{- $alt := $accTitle | default $accDescr -}} {{- if not $alt -}} {{- warnf "Diagram '%s' op pagina %s heeft geen accTitle of accDescr; geef het een tekstalternatief (WCAG 1.1.1)." $slug .Page.File.Path -}} {{- $alt = printf "Diagram %d" $blok.nummer -}} {{- end -}} {{- $beschrijving := "" -}} {{- if $accTitle -}}{{ $beschrijving = $accDescr }}{{- end -}} {{- $kB := int (math.Ceil (div (float (len $lightSVG)) 1024)) -}}
