{{- /* Schoont RawContent op en herschrijft interne links naar hun .md-variant. Aanroepen met de oorsprongpagina als context (niet met een string): .GetPage resolvet relatieve paden t.o.v. die pagina. Wie de tabellen ongemoeid wil laten, roept aan met (dict "page" . "tabellen" true). Met "html" true wijzen interne links naar de absolute URL van de HTML-pagina, voor losse documenten buiten de site. */ -}} {{- $p := . -}} {{- $tabellen := false -}} {{- $html := false -}} {{- if reflect.IsMap . -}} {{- $p = .page -}} {{- $tabellen = .tabellen -}} {{- $html = .html -}} {{- end -}} {{- $content := $p.RawContent -}} {{- $content = partial "markdown-extra.html" (dict "tekst" $content "page" $p) -}} {{- /* GFM kent geen tabel zonder koprij. In de gepubliceerde .md wordt table-without-header daarom een lijst; pandoc krijgt een lege koprij, want zonder koprij ziet die geen tabel. Moet vóór het strippen van de shortcodetags. */ -}} {{- range findRE `\{\{<\s*table-without-header\s*>\}\}[\s\S]*?\{\{<\s*/\s*table-without-header\s*>\}\}` $content -}} {{- $rijen := slice -}} {{- $koprij := "" -}} {{- range split . "\n" -}} {{- $regel := trim . " " -}} {{- if hasPrefix $regel "|" -}} {{- if findRE `^\|[\s:|-]+\|$` $regel -}} {{- $koprij = index (last 1 $rijen) 0 -}} {{- else -}} {{- $rijen = $rijen | append $regel -}} {{- end -}} {{- end -}} {{- end -}} {{- if $koprij }}{{ $rijen = after 1 $rijen }}{{ end -}} {{- $regels := slice -}} {{- if $tabellen -}} {{- $leeg := slice -}} {{- $streep := slice -}} {{- range seq (len (split (trim (index $rijen 0) "|") "|")) -}} {{- $leeg = $leeg | append " " -}} {{- $streep = $streep | append " --- " -}} {{- end -}} {{- $regels = $regels | append (printf "|%s|" (delimit $leeg "|")) (printf "|%s|" (delimit $streep "|")) -}} {{- range $rijen }}{{ $regels = $regels | append . }}{{ end -}} {{- else -}} {{- range $rijen -}} {{- $cellen := apply (split (trim . "|") "|") "trim" "." " " -}} {{- $regels = $regels | append (printf "- **%s:** %s" (index $cellen 0) (delimit (after 1 $cellen) ", ")) -}} {{- end -}} {{- end -}} {{- $content = replace $content . (delimit $regels "\n") -}} {{- end -}} {{- /* Moet vóór het strippen hieronder, anders verdwijnt de emoji mee. */ -}} {{- $content = replaceRE `\{\{<\s*emoji\s[^>]*?teken="([^"]+)"[^>]*>\}\}` "$1" $content -}} {{- $content = replaceRE `\{\{<[^>]+>\}\}` "" $content -}} {{- $content = replaceRE `\{\{%[^%]+%\}\}` "" $content -}} {{- /* Regel voor regel, met de binnenkant van codeblokken met hekken ongemoeid: een alert wordt een citaat met de typenaam vet vooraan, {ref:N} en de attribuutlijsten van Goldmark ({#id}, {.klasse}, {naam="..."}) vallen weg. De typenamen zijn die van render-blockquote.html; note en info hebben er geen. */ -}} {{- $typenamen := dict "tip" "Tip" "important" "Belangrijk" "warning" "Waarschuwing" "caution" "Pas op" -}} {{- $attributen := `\{[ \t]*(?:[#.][^{}\n]*|[A-Za-z_:][\w:.-]*[ \t]*=[^{}\n]*)\}` -}} {{- $losseAttributen := printf `^[ \t>]*%s[ \t]*$` $attributen -}} {{- $kopAttributen := printf `^([ \t>]*#{1,6}[ \t].*?)[ \t]*%s[ \t]*$` $attributen -}} {{- $regels := split $content "\n" -}} {{- $uit := slice -}} {{- $hek := "" -}} {{- range $i, $regel := $regels -}} {{- $streep := "" -}} {{- with findRESubmatch `^[ \t>]*(\x60{3,}|~{3,})` $regel 1 }}{{ $streep = index . 0 1 }}{{ end -}} {{- if $hek -}} {{- /* Het blok sluit met alleen hetzelfde teken, minstens zo vaak. */ -}} {{- if and $streep (eq (trim $regel " \t>") $streep) (hasPrefix $streep $hek) }}{{ $hek = "" }}{{ end -}} {{- else if $streep -}} {{- $hek = $streep -}} {{- else -}} {{- $regel = replaceRE `[ \t]*\{ref:\d+\}` "" $regel -}} {{- if findRE $losseAttributen $regel 1 }}{{ continue }}{{ end -}} {{- $regel = replaceRE $kopAttributen "$1" $regel -}} {{- /* Alleen de eerste regel van een citaat kan een alert openen. */ -}} {{- $eerste := true -}} {{- if gt $i 0 }}{{ $eerste = not (findRE `^[ \t]*>` (index $regels (sub $i 1)) 1) }}{{ end -}} {{- with and $eerste (findRESubmatch `^([ \t]*>[ \t>]*)\[!([A-Za-z]+)\][+-]?[ \t]*(.*?)[ \t]*$` $regel 1) -}} {{- $prefix := index . 0 1 -}} {{- $titel := index . 0 3 -}} {{- $kop := "" -}} {{- with index $typenamen (lower (index . 0 2)) }}{{ $kop = printf "**%s:**" . }}{{ end -}} {{- with $titel }}{{ $kop = cond (eq $kop "") (printf "**%s**" .) (printf "%s %s" $kop .) }}{{ end -}} {{- if not $kop }}{{ continue }}{{ end -}} {{- $regel = printf "%s%s" $prefix $kop -}} {{- /* Na een titel begint de tekst een eigen alinea, zoals onder een kop. */ -}} {{- if and $titel (lt (add $i 1) (len $regels)) (findRE `^[ \t]*>[ \t>]*\S` (index $regels (add $i 1)) 1) -}} {{- $uit = $uit | append $regel -}} {{- $regel = strings.TrimRight " \t" $prefix -}} {{- end -}} {{- end -}} {{- end -}} {{- $uit = $uit | append $regel -}} {{- end -}} {{- $content = delimit $uit "\n" -}} {{- /* Word en Windows kappen een hyperlink af op 2083 tekens en melden het document daarna als beschadigd. In de uitvoer voor buiten de site wijst zo'n link daarom naar de webpagina, waar hij wel werkt. */ -}} {{- if $html -}} {{- range findRESubmatch `\[([^\]]*)\]\((https?://[^)\s]+)\)` $content -}} {{- $geheel := index . 0 -}} {{- if gt (len (index . 2)) 2083 -}} {{- $content = replace $content $geheel (printf "%s (deze URL is te lang voor dit document, maar op [de site van %s](%s) vind je die wel)" (index . 1) $p.Site.Title $p.Permalink) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* findRESubmatch-groepen: 0=geheel, 1='!', 2=tekst, 3=doel */ -}} {{- range findRESubmatch `(!?)\[([^\]]*)\]\(([^)\s]+)\)` $content -}} {{- $whole := index . 0 -}} {{- $text := index . 2 -}} {{- $dest := index . 3 -}} {{- if eq (index . 1) "!" }}{{ continue }}{{ end -}} {{- if hasPrefix $dest "<" }}{{ continue }}{{ end -}} {{- $u := urls.Parse $dest -}} {{- if or $u.IsAbs (hasPrefix $dest "#") (hasPrefix $dest "//") (hasPrefix $dest "mailto:") (hasPrefix $dest "tel:") (not $u.Path) }}{{ continue }}{{ end -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $new := "" -}} {{- $target := $p.GetPage $path -}} {{- with $target -}} {{- $isMd := false -}} {{- with .File }}{{ if eq .Ext "md" }}{{ $isMd = true }}{{ end }}{{ end -}} {{- $mdOut := .OutputFormats.Get "markdown" -}} {{- if $html -}} {{- $new = .Permalink -}} {{- else if and $isMd $mdOut -}} {{- $new = $mdOut.RelPermalink -}} {{- else -}} {{- $new = .RelPermalink -}} {{- end -}} {{- else -}} {{- with or ($p.Resources.Get $path) (resources.Get $path) -}} {{- $new = cond $html .Permalink .RelPermalink -}} {{- end -}} {{- end -}} {{- /* Bijvoorbeeld een alias: GetPage vindt die niet, de site stuurt wel door. */ -}} {{- if and $html (not $new) (hasPrefix $path "/") }}{{ $new = absURL $path }}{{ end -}} {{- if $new -}} {{- with $u.RawQuery }}{{ $new = printf "%s?%s" $new . }}{{ end -}} {{- with $u.Fragment }}{{ $new = printf "%s#%s" $new . }}{{ end -}} {{- $content = replace $content $whole (printf "[%s](%s)" $text $new) -}} {{- end -}} {{- end -}} {{- $content = replaceRE `\n{3,}` "\n\n" $content -}} {{- return ($content | strings.TrimSpace | safeHTML) -}}
