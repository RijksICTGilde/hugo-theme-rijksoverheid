{{- /* Render tekst met {ref:N} → onderstippelde underline op het direct voorafgaande woord. Hover/focus toont een tooltip met references[N-1].text en (indien aanwezig) een Bekijk-bron-link. Gebruik: {{ partial "reftext.html" (dict "text" .tekst "page" $page) }} */ -}} {{- if and (isset . "tekst") (not (isset . "text")) -}} {{- with .page -}} {{- warnf "%s: reftext.html krijgt tekst; die sleutel heet nu text." .Path -}} {{- else -}} {{- warnf "reftext.html krijgt tekst; die sleutel heet nu text." -}} {{- end -}} {{- end -}} {{- $text := .page.RenderString .text -}} {{- $base := "x" | relURL -}} {{- $base = strings.TrimSuffix "x" $base -}} {{- if ne $base "/" -}} {{- $text = replace $text "href=\"/" (printf "href=\"%s" $base) -}} {{- end -}} {{- $page := .page -}} {{- range $i, $ref := $page.Params.references -}} {{- $num := add $i 1 -}} {{- /* replaceRE leest een $ in de vervanging als groep: "$5" en "?a=$1" moeten letterlijk blijven, dus $$. */ -}} {{- $textEsc := replace ($ref.text | htmlEscape) "$" "$$" -}} {{- $linkHTML := "" -}} {{- with $ref.url -}} {{- $linkHTML = replace (partial "render-link.html" (dict "href" . "text" "Bekijk bron")) "$" "$$" -}} {{- end -}} {{- $pattern := printf "([^\\s<>]*)(\\s*)\\{ref:%d\\}" $num -}} {{- /* De teller per bron staat op de pagina: elk voorkomen krijgt een eigen id, ook als dit partial vaker op een pagina draait, en alleen het eerste is het anker voor de ↩-link onderaan (refback-N). Hugo server bewaart .Page.Store na een templatewijziging; de teller loopt dan door en dat anker ontbreekt tot een herstart. */ -}} {{- $key := printf "reftext-%d" $num -}} {{- range findRE $pattern $text -}} {{- $page.Store.Add $key 1 -}} {{- $sequenceNumber := $page.Store.Get $key -}} {{- $tipId := printf "ref-tip-%d-%d" $num $sequenceNumber -}} {{- $anchor := cond (eq $sequenceNumber 1) (printf " id=\"refback-%d\"" $num) "" -}} {{- /* Zonder woord ervoor (na een tag of een andere {ref:N}) is het nummer de term; een lege knop heeft geen naam en is niet te zien. */ -}} {{- $term := "${1}" -}} {{- $name := "" -}} {{- if not (index (findRESubmatch $pattern $text 1) 0 1) -}} {{- $term = printf "%d" $num -}} {{- $name = printf " aria-label=\"Bron %d\"" $num -}} {{- end -}} {{- $replace := printf "%s%s${2}" $anchor $name $tipId $term $tipId $textEsc $linkHTML -}} {{- $text = replaceRE $pattern $replace $text 1 -}} {{- end -}} {{- end -}} {{- $text | safeHTML -}}
