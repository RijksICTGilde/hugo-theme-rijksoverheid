{{- /* Render tekst met {ref:N} → onderstippelde underline op het direct voorafgaande woord. Hover/focus toont een tooltip met referenties[N-1].tekst en (indien aanwezig) een Bekijk-bron-link. Gebruik: {{ partial "reftext.html" (dict "tekst" .tekst "page" $page) }} */ -}} {{- $tekst := .tekst | markdownify -}} {{- $base := "x" | relURL -}} {{- $base = strings.TrimSuffix "x" $base -}} {{- if ne $base "/" -}} {{- $tekst = replace $tekst "href=\"/" (printf "href=\"%s" $base) -}} {{- end -}} {{- $page := .page -}} {{- range $i, $ref := $page.Params.referenties -}} {{- $num := add $i 1 -}} {{- /* replaceRE leest een $ in de vervanging als groep: "$5" en "?a=$1" moeten letterlijk blijven, dus $$. */ -}} {{- $textEsc := replace ($ref.tekst | htmlEscape) "$" "$$" -}} {{- $linkHTML := "" -}} {{- with $ref.url -}} {{- $linkHTML = replace (partial "render-link.html" (dict "href" . "text" "Bekijk bron")) "$" "$$" -}} {{- end -}} {{- $pattern := printf "([^\\s<>]*)(\\s*)\\{ref:%d\\}" $num -}} {{- /* De teller per bron staat op de pagina: elk voorkomen krijgt een eigen id, ook als dit partial vaker op een pagina draait, en alleen het eerste is het anker voor de ↩-link onderaan (refback-N). Hugo server bewaart .Page.Store na een templatewijziging; de teller loopt dan door en dat anker ontbreekt tot een herstart. */ -}} {{- $sleutel := printf "reftext-%d" $num -}} {{- range findRE $pattern $tekst -}} {{- $page.Store.Add $sleutel 1 -}} {{- $volgnummer := $page.Store.Get $sleutel -}} {{- $tipId := printf "ref-tip-%d-%d" $num $volgnummer -}} {{- $anker := cond (eq $volgnummer 1) (printf " id=\"refback-%d\"" $num) "" -}} {{- /* Zonder woord ervoor (na een tag of een andere {ref:N}) is het nummer de term; een lege knop heeft geen naam en is niet te zien. */ -}} {{- $term := "${1}" -}} {{- $naam := "" -}} {{- if not (index (findRESubmatch $pattern $tekst 1) 0 1) -}} {{- $term = printf "%d" $num -}} {{- $naam = printf " aria-label=\"Bron %d\"" $num -}} {{- end -}} {{- $replace := printf "%s%s${2}" $anker $naam $tipId $term $tipId $textEsc $linkHTML -}} {{- $tekst = replaceRE $pattern $replace $tekst 1 -}} {{- end -}} {{- end -}} {{- $tekst | safeHTML -}}
