{{- /* Emoji met een bewuste keuze voor schermlezers. Een schermlezer leest een kale emoji voor met een eigen naam, vaak Engels of per apparaat anders. Kies daarom altijd een van de twee: {{< emoji symbol="🚫" aria-label="Kan beter" >}} betekenisvol: leest "Kan beter" {{< emoji symbol="💡" aria-hidden=true >}} versiering: leest niets In de Markdown-uitvoer wordt het de kale emoji (partial markdown-body). */ -}} {{- if not .IsNamedParams -}} {{- errorf "%s: emoji kent alleen benoemde parameters, bijvoorbeeld symbol=\"💡\"" .Position -}} {{- else -}} {{- range $name, $_ := .Params -}} {{- if not (in (slice "symbol" "aria-label" "aria-hidden") $name) -}} {{- errorf "%s: onbekende parameter %q voor emoji; kies uit symbol, aria-label, aria-hidden" $.Position $name -}} {{- end -}} {{- end -}} {{- $symbol := .Get "symbol" -}} {{- $label := .Get "aria-label" -}} {{- $hidden := false -}} {{- with .Get "aria-hidden" -}} {{- if not (in (slice "true" "false") (string .)) -}} {{- errorf "%s: aria-hidden is true of false, niet %q" $.Position (string .) -}} {{- end -}} {{- $hidden = eq (string .) "true" -}} {{- end -}} {{- if not $symbol -}} {{- errorf "%s: emoji mist de parameter symbol" .Position -}} {{- end -}} {{- if and $label $hidden -}} {{- errorf "%s: emoji heeft aria-label of aria-hidden=true, niet allebei" .Position -}} {{- end -}} {{- if not (or $label $hidden) -}} {{- errorf "%s: emoji mist een keuze voor schermlezers: geef aria-label=\"..\" (betekenisvol) of aria-hidden=true (versiering)" .Position -}} {{- end -}} {{- if $hidden -}} {{ $symbol }} {{- else -}} {{ $symbol }} {{- end -}} {{- end -}}
