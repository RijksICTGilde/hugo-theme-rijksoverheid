{{- /* Tabel zonder koprij, met een willekeurig aantal kolommen. De eerste kolom wordt de rijkop, want een tabel zonder enige kop is voor een schermlezer betekenisloos. De inhoud is één rij per regel, met pijpen als scheiding, net als in een Markdown-tabel maar zonder koprij en zonder scheidingsregel. Staat er toch een scheidingsregel, dan wordt die met de koprij erboven genegeerd. print-bijeen="ja" houdt de tabel in de afdruk op één pagina, net als het gelijknamige attribuut op een Markdown-tabel. Zonder label=".." heet de wikkel "Tabel"; inhoud.html maakt daar "Tabel 1", "Tabel 2" van, zodat elke regio een eigen naam heeft. */ -}} {{- $regels := slice -}} {{- $koprij := "" -}} {{- range split (trim .Inner "\n ") "\n" -}} {{- $regel := trim . " " -}} {{- if hasPrefix $regel "|" -}} {{- if findRE `^\|[\s:|-]+\|$` $regel -}} {{- if not $regels -}} {{- errorf "table-without-header: de scheidingsregel `%s` staat op de eerste regel; zet er een koprij boven of haal hem weg, op pagina %s" $regel $.Page.File.Path -}} {{- end -}} {{- $koprij = index (last 1 $regels) 0 -}} {{- else -}} {{- $regels = $regels | append $regel -}} {{- end -}} {{- end -}} {{- end -}} {{- if $koprij -}} {{- $regels = after 1 $regels -}} {{- end -}} {{- if not $regels -}} {{- errorf "table-without-header: verwacht rijen in de vorm `| a | b |`, op pagina %s" .Page.File.Path -}} {{- end -}}
{{- range $regels }} {{- $cellen := split (trim . "|") "|" }} {{ trim (index $cellen 0) " " | markdownify }} {{- range after 1 $cellen }} {{ trim . " " | markdownify }} {{- end }} {{- end }}
